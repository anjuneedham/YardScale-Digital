# yardscaledigital.com — Apache config for Namecheap shared hosting (cPanel)
#
# Every directive here is either guarded by <IfModule> or is core Apache, so a
# missing module degrades quietly instead of returning 500.
#
# NOTE ON HTTPS: forcing https from .htaccess is deliberately NOT done here.
# When the host terminates SSL upstream and doesn't pass X-Forwarded-Proto, the
# usual rewrite sees every request as plain http and redirects forever
# (ERR_TOO_MANY_REDIRECTS). Use cPanel's own switch instead — it knows how the
# server is fronted:  cPanel -> Domains -> Force HTTPS Redirect.

Options -Indexes
DirectoryIndex index.html index.htm

ErrorDocument 404 /404.html

# ─────────── tidy URLs ───────────
<IfModule mod_rewrite.c>
  RewriteEngine On

  # www -> bare domain. REQUEST_SCHEME keeps whatever protocol they arrived on,
  # so this still works before the SSL certificate has been issued.
  RewriteCond %{HTTP_HOST} ^www\.(.+)$ [NC]
  RewriteRule ^ %{REQUEST_SCHEME}://%1%{REQUEST_URI} [R=301,L]

  # /index.html -> /
  RewriteCond %{THE_REQUEST} \s/+index\.html[\s?] [NC]
  RewriteRule ^ / [R=301,L]

  # /thanks -> /thanks.html  (internal, no redirect)
  RewriteCond %{REQUEST_FILENAME} !-f
  RewriteCond %{REQUEST_FILENAME} !-d
  RewriteRule ^([^./]+)/?$ $1.html [L]
</IfModule>

# ─────────── security headers ───────────
<IfModule mod_headers.c>
  Header always set X-Content-Type-Options "nosniff"
  Header always set X-Frame-Options "SAMEORIGIN"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
  Header always set Permissions-Policy "geolocation=(), microphone=(), camera=()"
  # Only switch this on once https works everywhere — it commits browsers to
  # https for a year and is painful to undo.
  # Header always set Strict-Transport-Security "max-age=31536000; includeSubDomains"
</IfModule>

# ─────────── compression ───────────
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/plain text/css text/xml
  AddOutputFilterByType DEFLATE application/javascript application/json
  AddOutputFilterByType DEFLATE application/xml image/svg+xml
</IfModule>

# ─────────── caching ───────────
# HTML always revalidates, so an edit you upload is live on refresh.
# Images and icons are content-stable, so cache them hard.
<IfModule mod_expires.c>
  ExpiresActive On
  ExpiresDefault                        "access plus 1 month"
  ExpiresByType text/html               "access plus 0 seconds"
  ExpiresByType image/png               "access plus 1 year"
  ExpiresByType image/jpeg              "access plus 1 year"
  ExpiresByType image/svg+xml           "access plus 1 year"
  ExpiresByType image/x-icon            "access plus 1 year"
  ExpiresByType text/css                "access plus 1 year"
  ExpiresByType application/javascript  "access plus 1 year"
</IfModule>

<IfModule mod_headers.c>
  <FilesMatch "\.html$">
    Header set Cache-Control "public, max-age=0, must-revalidate"
  </FilesMatch>
</IfModule>

# ─────────── don't serve the notes ───────────
# Apache 2.4 and 2.2 spell this differently; using the wrong one is a 500.
<IfModule mod_authz_core.c>
  <FilesMatch "\.(md|sh|toml)$">
    Require all denied
  </FilesMatch>
</IfModule>
<IfModule !mod_authz_core.c>
  <FilesMatch "\.(md|sh|toml)$">
    Order allow,deny
    Deny from all
  </FilesMatch>
</IfModule>
